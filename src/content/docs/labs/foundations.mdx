```python
"""L01: inspect a real, small pretrained decoder and its continuation."""

import json
from pathlib import Path
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer

ROOT = Path(__file__).resolve().parents[6]
MODEL = "EleutherAI/pythia-14m"
REVISION = "cf967c0a9a04383db6f7b1108d86b2962634b4ac"


def main():
    torch.manual_seed(11)
    torch.set_num_threads(1)
    tokenizer = AutoTokenizer.from_pretrained(MODEL, revision=REVISION)
    model = AutoModelForCausalLM.from_pretrained(
        MODEL, revision=REVISION, attn_implementation="eager"
    ).eval()
    prompt = "The sun rises in the"
    encoded = tokenizer(prompt, return_tensors="pt")
    with torch.inference_mode():
        output = model(**encoded, output_hidden_states=True)
        generated = model.generate(
            **encoded,
            max_new_tokens=16,
            do_sample=False,
            pad_token_id=tokenizer.eos_token_id,
        )
    continuation = generated[0, encoded.input_ids.shape[1] :]
    assert torch.equal(generated[0, : encoded.input_ids.shape[1]], encoded.input_ids[0])
    hidden = output.hidden_states[-1]
    with torch.inference_mode():
        projected = model.get_output_embeddings()(hidden)
    assert torch.allclose(projected, output.logits, atol=1e-5)
    result = {
        "prompt": prompt,
        "input_ids": encoded.input_ids[0].tolist(),
        "pieces": tokenizer.convert_ids_to_tokens(encoded.input_ids[0].tolist()),
        "new_ids": continuation.tolist(),
        "continuation": tokenizer.decode(continuation, skip_special_tokens=True),
        "input_shape": list(encoded.input_ids.shape),
        "embedding_shape": list(model.get_input_embeddings().weight.shape),
        "hidden_shape": list(hidden.shape),
        "logits_shape": list(output.logits.shape),
        "layers": model.config.num_hidden_layers,
        "heads": model.config.num_attention_heads,
        "parameters": sum(p.numel() for p in model.parameters()),
        "prompt_preserved": True,
        "lm_head_checked": True,
    }
    path = ROOT / "examples/artifacts/e3-generation"
    path.mkdir(parents=True, exist_ok=True)
    (path / "results.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(json.dumps(result))


if __name__ == "__main__":
    main()
```

```python
"""L03: pretrained static words, contextual tokens, and pooled sentences."""

import gzip
import hashlib
import json
import os
from pathlib import Path
import urllib.request
import numpy as np
import torch
from transformers import AutoModel, AutoTokenizer

ROOT = Path(__file__).resolve().parents[6]
GLOVE_URL = "https://github.com/RaRe-Technologies/gensim-data/releases/download/glove-wiki-gigaword-50/glove-wiki-gigaword-50.gz"
GLOVE_SHA = "5c55f98957aa9fed8d2ac5fb1dcff57af3b23c5a3ee7af3f7945f8d49198eb24"
BERT = "prajjwal1/bert-tiny"
BERT_REV = "6f75de8b60a9f8a2fdf7b69cbd86d9e64bcb3837"
TOK = "google-bert/bert-base-uncased"
TOK_REV = "86b5e0934494bd15c9632b12f734a8a67f723594"
SENTENCE = "sentence-transformers/paraphrase-MiniLM-L3-v2"
SENTENCE_REV = "4ca70771034acceecb2e72475f72050fcdde4ddc"


def main():
    torch.set_num_threads(1)
    path = ROOT / ".cache/e3/glove-wiki-gigaword-50.gz"
    path.parent.mkdir(parents=True, exist_ok=True)
    if not path.exists():
        if os.getenv("HF_HUB_OFFLINE") == "1":
            raise RuntimeError(
                "GloVe cache missing: rerun explicitly with --allow-network"
            )
        urllib.request.urlretrieve(GLOVE_URL, path)
    assert hashlib.sha256(path.read_bytes()).hexdigest() == GLOVE_SHA
    words = []
    with gzip.open(path, "rt", encoding="utf8") as stream:
        rows, width = map(int, next(stream).split())
        vectors = np.empty((rows, width), dtype=np.float32)
        for i, line in enumerate(stream):
            fields = line.split()
            words.append(fields[0])
            vectors[i] = np.asarray(fields[1:], dtype=np.float32)
    index = {word: i for i, word in enumerate(words)}
    vectors /= np.maximum(np.linalg.norm(vectors, axis=1, keepdims=True), 1e-12)
    neighbors = {}
    for query in ["king", "bank", "good"]:
        scores = vectors @ vectors[index[query]]
        scores[index[query]] = -np.inf
        neighbors[query] = [
            {"word": words[i], "cosine": round(float(scores[i]), 5)}
            for i in np.argsort(-scores)[:5]
        ]
    tokenizer = AutoTokenizer.from_pretrained(TOK, revision=TOK_REV, use_fast=True)
    model = AutoModel.from_pretrained(BERT, revision=BERT_REV).eval()
    sentences = ["The bank approved the loan.", "The bank flooded after rain."]
    encoded = tokenizer(sentences, padding=True, return_tensors="pt")
    with torch.inference_mode():
        lookup = model.get_input_embeddings()(encoded.input_ids)
        hidden = model(**encoded).last_hidden_state
    positions = [
        ids.index(tokenizer.convert_tokens_to_ids("bank"))
        for ids in encoded.input_ids.tolist()
    ]
    a, b = [hidden[i, pos] for i, pos in enumerate(positions)]
    assert torch.equal(lookup[0, positions[0]], lookup[1, positions[1]])
    assert not torch.allclose(a, b)
    contextual_cosine = float(
        torch.nn.functional.cosine_similarity(a[None], b[None])[0]
    )
    st = AutoTokenizer.from_pretrained(SENTENCE, revision=SENTENCE_REV)
    sm = AutoModel.from_pretrained(SENTENCE, revision=SENTENCE_REV).eval()
    texts = sentences + ["A lender accepted a credit application."]
    batch = st(texts, padding=True, return_tensors="pt")
    with torch.inference_mode():
        states = sm(**batch).last_hidden_state
        mask = batch.attention_mask.unsqueeze(-1)
        pooled = (states * mask).sum(1) / mask.sum(1)
        pooled = torch.nn.functional.normalize(pooled, dim=1)
    similarities = (pooled @ pooled.T).tolist()
    selected = ["king", "queen", "bank", "river", "loan", "good", "bad", "excellent"]
    small = vectors[[index[w] for w in selected]]
    centered = small - small.mean(0)
    _, singular, rotation = np.linalg.svd(centered, full_matrices=False)
    projection = centered @ rotation[:2].T
    result = {
        "static_shape": [rows, width],
        "neighbors": neighbors,
        "bank_token_id": tokenizer.convert_tokens_to_ids("bank"),
        "lookup_shape": list(lookup.shape),
        "hidden_shape": list(hidden.shape),
        "same_lookup": True,
        "different_context": True,
        "bank_context_cosine": contextual_cosine,
        "sentence_shape": list(pooled.shape),
        "sentence_cosines": similarities,
        "sentences": texts,
        "projection": {
            "words": selected,
            "coordinates": projection.tolist(),
            "variance_fraction": float((singular[:2] ** 2).sum() / (singular**2).sum()),
            "method": "PCA of eight selected normalized 50D GloVe rows; nearest neighbors computed in 50D, not this projection",
        },
        "numerical_token_counts": {
            name: len(tok("12.0*50=600\n\t42!", add_special_tokens=True).input_ids)
            for name, tok in [("bert", tokenizer), ("sentence", st)]
        },
        "glove_sha256": GLOVE_SHA,
    }
    out = ROOT / "examples/artifacts/e3-representations"
    out.mkdir(parents=True, exist_ok=True)
    (out / "results.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(json.dumps(result))


if __name__ == "__main__":
    main()
```

## L04: learned playlist recommendations

**Question:** do learned co-occurrences improve held-out ranking relative to popularity in this fixed fixture?

```bash
python3 scripts/run-examples.py --id playlist-embeddings
```

The original fictional catalog has 21 warm items plus one cold-start item. Forty-eight training sessions fit input/context tables using a radius-two skip-gram window, four sampled negatives per pair, and 180 CPU optimizer steps. Nine validation sessions provide a development view; the fixed procedure is evaluated on twelve final test sessions. Entire sessions and their unordered song sets are distinct across the splits. The catalog supplies IDs; no held-out interactions enter pair fitting or popularity counts.

For each held-out session, hide its last song, pool vectors from the observed songs, exclude those observed songs from candidates, and rank remaining warm items. Report Recall@3 and reciprocal rank with the query denominator. Negative sampling can include plausible co-occurrences; it is an approximation to a training objective, not a set of logically false musical relationships.

```python
"""L04: learn song vectors with skip-gram negative sampling, then evaluate."""

import json
import argparse
from collections import Counter
from pathlib import Path
import torch
from torch.nn import functional as F

ROOT = Path(__file__).resolve().parents[6]


def main(argv=None):
    torch.manual_seed(11)
    torch.set_num_threads(1)
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--data", type=Path, default=ROOT / "examples/data/fixtures/playlists.json"
    )
    args = parser.parse_args(argv)
    data = json.loads(args.data.read_text())
    catalog = [row["id"] for row in data["items"]]
    index = {name: i for i, name in enumerate(catalog)}
    train = [row for row in data["playlists"] if row["split"] == "train"]
    counts = Counter(song for row in train for song in row["songs"])
    active = [index[song] for song in sorted(counts)]
    pairs = []
    for row in train:
        for i, center in enumerate(row["songs"]):
            for j in range(max(0, i - 2), min(len(row["songs"]), i + 3)):
                if i != j:
                    pairs.append((index[center], index[row["songs"][j]]))
    pairs = torch.tensor(pairs)
    center = torch.nn.Embedding(len(catalog), 8)
    context = torch.nn.Embedding(len(catalog), 8)
    torch.nn.init.normal_(center.weight, std=0.1)
    torch.nn.init.zeros_(context.weight)
    optimizer = torch.optim.Adam(
        list(center.parameters()) + list(context.parameters()), lr=0.035
    )
    noise = torch.tensor(
        [counts[catalog[i]] ** 0.75 for i in range(len(catalog))], dtype=torch.float
    )
    original = center.weight.detach().clone()
    losses = []
    for step in range(180):
        optimizer.zero_grad()
        batch = pairs[torch.randint(len(pairs), (64,))]
        u = center(batch[:, 0])
        positive = context(batch[:, 1])
        negatives = torch.multinomial(noise, 64 * 4, replacement=True).reshape(64, 4)
        negative_vectors = context(negatives)
        objective = (
            F.softplus(-(u * positive).sum(-1)).mean()
            + F.softplus((u[:, None, :] * negative_vectors).sum(-1)).sum(-1).mean()
        )
        objective.backward()
        optimizer.step()
        if step % 30 == 0:
            losses.append(float(objective.detach()))
    assert not torch.equal(original, center.weight.detach())
    vectors = F.normalize(center.weight.detach(), dim=1)

    def evaluate(split):
        cases = []
        for row in data["playlists"]:
            if row["split"] != split:
                continue
            observed, target = row["songs"][:-1], row["songs"][-1]
            if target not in counts or any(song not in counts for song in observed):
                continue
            candidates = [song for song in counts if song not in observed]
            query = F.normalize(vectors[[index[s] for s in observed]].mean(0), dim=0)
            learned = sorted(
                candidates, key=lambda s: (-float(vectors[index[s]] @ query), s)
            )
            popular = sorted(candidates, key=lambda s: (-counts[s], s))
            cases.append(
                {
                    "id": row["id"],
                    "observed": observed,
                    "target": target,
                    "learned": learned[:3],
                    "popular": popular[:3],
                    "rank": learned.index(target) + 1,
                    "popular_rank": popular.index(target) + 1,
                }
            )
        if not cases:
            raise ValueError(
                "No warm-item evaluation queries; inspect user splits and coverage"
            )
        return {
            "queries": len(cases),
            "total_queries": sum(row["split"] == split for row in data["playlists"]),
            "cold_skipped": sum(row["split"] == split for row in data["playlists"])
            - len(cases),
            "recall_at_3": sum(r["target"] in r["learned"] for r in cases) / len(cases),
            "mrr": sum(1 / r["rank"] for r in cases) / len(cases),
            "popularity_recall_at_3": sum(r["target"] in r["popular"] for r in cases)
            / len(cases),
            "popularity_mrr": sum(1 / r["popular_rank"] for r in cases) / len(cases),
            "cases": cases,
        }

    assert "cold" not in counts
    for split in ["validation", "test"]:
        if not any(r["split"] == split for r in data["playlists"]):
            raise ValueError("Dataset needs train/validation/test sessions")
    path = ROOT / "examples/artifacts/e3-playlists"
    path.mkdir(parents=True, exist_ok=True)
    torch.save({"vectors": vectors, "catalog": catalog}, path / "vectors.pt")
    reloaded = torch.load(path / "vectors.pt", weights_only=True)
    assert torch.equal(reloaded["vectors"], vectors)
    result = {
        "validation": evaluate("validation"),
        "test": evaluate("test"),
        "losses": losses,
        "trained_pairs": len(pairs),
        "item_count": len(active),
        "cold_start": "cold has no training interactions; do not recommend its random row",
        "vectors": vectors[active].tolist(),
        "catalog": [catalog[i] for i in active],
        "learned_vectors": True,
        "reload_checked": True,
    }
    (path / "results.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(
        json.dumps({k: v for k, v in result.items() if k not in ["vectors", "catalog"]})
    )


if __name__ == "__main__":
    main()
```

## L05: transparent transformer

```python
"""L05: explicit random-weight transformer, checked against PyTorch attention."""

import json
from pathlib import Path
import numpy as np
import torch
from torch.nn import functional as F

ROOT = Path(__file__).resolve().parents[6]


def main():
    torch.set_num_threads(1)
    rng = np.random.default_rng(11)
    ids = np.array([0, 1, 2])
    embedding = rng.normal(size=(7, 4)) * 0.2
    x = embedding[ids]  # [T=3, d_model=4]
    norm = lambda z: (
        (z - z.mean(-1, keepdims=True)) / np.sqrt(z.var(-1, keepdims=True) + 1e-5)
    )
    normalized = norm(x)
    Wq, Wk, Wv, Wo = [rng.normal(size=(4, 4)) * 0.2 for _ in range(4)]
    split = lambda z: z.reshape(3, 2, 2).transpose(1, 0, 2)
    q, k, v = [split(normalized @ matrix) for matrix in [Wq, Wk, Wv]]
    scores = q @ k.transpose(0, 2, 1) / np.sqrt(2)
    mask = np.tril(np.ones((3, 3), dtype=bool))
    scores = np.where(mask, scores, -np.inf)
    exponent = np.exp(scores - scores.max(-1, keepdims=True))
    weights = exponent / exponent.sum(-1, keepdims=True)
    attended = weights @ v
    reference = F.scaled_dot_product_attention(
        *[torch.tensor(z) for z in [q, k, v]], is_causal=True
    )
    error = float(np.abs(attended - reference.numpy()).max())
    assert error < 1e-10
    assert np.all(weights[:, ~mask] == 0) and np.allclose(weights.sum(-1), 1.0)
    concatenated = attended.transpose(1, 0, 2).reshape(3, 4)
    residual = x + concatenated @ Wo
    W1 = rng.normal(size=(4, 8)) * 0.2
    W2 = rng.normal(size=(8, 4)) * 0.2
    hidden = residual + np.maximum(0, norm(residual) @ W1) @ W2
    logits = hidden @ rng.normal(size=(4, 7))
    # Changing a future input cannot change earlier masked outputs.
    changed_v = v.copy()
    changed_v[:, -1, :] += 100
    assert np.allclose((weights @ changed_v)[:, :2, :], attended[:, :2, :])
    result = {
        "input_shape": list(x.shape),
        "qkv_shape": list(q.shape),
        "attention_shape": list(weights.shape),
        "head_shape": list(logits.shape),
        "weights": weights[0].round(6).tolist(),
        "values": v[0].round(6).tolist(),
        "output": hidden.round(6).tolist(),
        "logits": logits[-1].round(6).tolist(),
        "reference_max_error": error,
        "future_masked": True,
        "future_perturbation_checked": True,
        "random_weights": True,
    }
    path = ROOT / "examples/artifacts/e3-transformer"
    path.mkdir(parents=True, exist_ok=True)
    (path / "results.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(json.dumps(result))


if __name__ == "__main__":
    main()
```

```python
"""L06: cached/uncached correctness before a small warmed CPU timing."""

import json
from pathlib import Path
from time import perf_counter
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer

ROOT = Path(__file__).resolve().parents[6]
MODEL = "EleutherAI/pythia-14m"
REVISION = "cf967c0a9a04383db6f7b1108d86b2962634b4ac"


def main():
    torch.set_num_threads(1)
    torch.manual_seed(11)
    tokenizer = AutoTokenizer.from_pretrained(MODEL, revision=REVISION)
    model = (
        AutoModelForCausalLM.from_pretrained(
            MODEL, revision=REVISION, attn_implementation="eager"
        )
        .double()
        .eval()
    )
    prompt = tokenizer("The sun rises in the", return_tensors="pt").input_ids
    forced = tokenizer(
        " east and the sky becomes bright.", add_special_tokens=False
    ).input_ids

    def run(cached):
        prefix = prompt.clone()
        values = []
        past = None
        out = None
        with torch.inference_mode():
            for step in range(len(forced) + 1):
                token_ids = prefix if not cached or step == 0 else prefix[:, -1:]
                out = model(
                    input_ids=token_ids,
                    past_key_values=past if cached else None,
                    attention_mask=torch.ones_like(prefix),
                    use_cache=cached,
                )
                values.append(out.logits[:, -1].clone())
                if cached:
                    past = out.past_key_values
                if step < len(forced):
                    prefix = torch.cat([prefix, torch.tensor([[forced[step]]])], dim=1)
        return values, past, prefix.shape[1]

    full, _, length = run(False)
    cached, past, _ = run(True)
    errors = [(a - b).abs().max().item() for a, b in zip(full, cached)]
    assert max(errors) < 1e-9, errors
    cache_layers = list(past)
    shapes = [list(key.shape) for key, value in cache_layers]
    assert all(shape[-2] == length for shape in shapes)
    measured_bytes = sum(
        t.numel() * t.element_size() for layer in cache_layers for t in layer
    )
    predicted_bytes = (
        2
        * model.config.num_hidden_layers
        * model.config.num_attention_heads
        * length
        * (model.config.hidden_size // model.config.num_attention_heads)
        * 8
    )
    assert measured_bytes == predicted_bytes
    timing = {}
    for mode in [False, True]:
        run(mode)  # warm-up, not included
        samples = []
        for _ in range(3):
            start = perf_counter()
            run(mode)
            samples.append(perf_counter() - start)
        timing["cached" if mode else "uncached"] = samples
    result = {
        "max_logit_error": max(errors),
        "cache_shape": shapes[0],
        "layers": len(shapes),
        "cache_bytes": measured_bytes,
        "positions": length,
        "forced_tokens": len(forced),
        "dtype": "float64",
        "cpu_seconds": timing,
        "correctness_checked": True,
        "memory_checked": True,
        "measurement": "CPU float64 correctness/timing demonstration, not a production-precision benchmark; warm-up then three same-token runs including prompt prefill. Tiny workloads can be slower with caching.",
    }
    path = ROOT / "examples/artifacts/e3-cache"
    path.mkdir(parents=True, exist_ok=True)
    (path / "results.json").write_text(json.dumps(result, indent=2, ensure_ascii=False) + "\n")
    print(json.dumps(result))


if __name__ == "__main__":
```
